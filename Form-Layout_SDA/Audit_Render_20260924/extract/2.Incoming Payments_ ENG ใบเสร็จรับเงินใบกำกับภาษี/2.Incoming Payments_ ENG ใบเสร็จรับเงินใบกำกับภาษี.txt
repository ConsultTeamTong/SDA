stringVar array Units := ["", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine", "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen", "Seventeen", "Eighteen", "Nineteen"];
stringVar array Tens := ["", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"];
stringVar array Scales := ["", "Thousand", "Million", "Billion"];

// แปลงค่าเงินเป็น String และจัดการทศนิยม
stringVar fullAmount := ToText({Bank_Receipt.DocTotalFC}, 2, ""); 
fullAmount := replace(fullAmount, ",", "");

stringVar array parts := Split(fullAmount, ".");
stringVar wholePart := parts[1];
stringVar decimalPart := parts[2];

// --- ส่วนของ Baht ---
stringVar result := "";
numberVar n := Val(wholePart);
numberVar i := 1;

if n = 0 then 
    result := "Zero"
else (
    while n > 0 do (
        numberVar chunk := n mod 1000;
        if chunk > 0 then (
            stringVar chunkText := "";
            numberVar h := Int(chunk / 100);
            numberVar t := chunk mod 100;
            
            if h > 0 then 
                chunkText := Units[h + 1] + " Hundred ";
            
            if t > 0 then (
                if t < 20 then 
                    chunkText := chunkText + Units[t + 1]
                else (
                    chunkText := chunkText + Tens[Int(t / 10) + 1];
                    if (t mod 10) > 0 then 
                        chunkText := chunkText + "-" + Units[(t mod 10) + 1];
                )
            );
            
            result := CStr(chunkText) + " " + CStr(Scales[i]) + " " + CStr(result);
        );
        n := Int(n / 1000);
        i := i + 1;
        result := result; // <-- เพิ่มบรรทัดนี้ เพื่อบังคับให้ Loop คืนค่าเป็น String
    );
    result; // <-- เพิ่มบรรทัดนี้ เพื่อให้ Block Else ทั้งหมดคืนค่าเป็น String
);

// --- ส่วนของ Satang ---
stringVar centsText := "";
numberVar cents := Val(decimalPart);
if cents > 0 then (
    if cents < 20 then 
        centsText := Units[cents + 1]
    else (
        centsText := Tens[Int(cents / 10) + 1];
        if (cents mod 10) > 0 then 
            centsText := centsText + "-" + Units[(cents mod 10) + 1];
    )
);

// --- สรุปผล ---
stringVar finalOutput := "";
finalOutput := Trim(result) + " Dollars";

if Trim(centsText) <> "" then
    finalOutput := finalOutput + " and " + Trim(centsText) + " cents"
else
    finalOutput := finalOutput + " Only";

// ป้องกัน Error กรณี String ว่างก่อนใช้ Uppercase
if Len(finalOutput) > 0 then
    Uppercase(Left(finalOutput, 1)) + Mid(finalOutput, 2)
else
    "";
