// =======================================================
// สูตรแปลงตัวเลขเป็นคำอ่านภาษาไทย (รองรับหลายสกุลเงิน)
// =======================================================
Local NumberVar InputNum := {AR_SQ.DocTotal}; // 1. เปลี่ยนฟิลด์ยอดเงินที่นี่
Local StringVar Cur := {AR_SQ.DocCur};          // 2. เปลี่ยนฟิลด์สกุลเงินที่นี่

// กำหนดชื่อหน่วยหลักและหน่วยย่อย
Local StringVar MajorUnit;
Local StringVar MinorUnit;

If Cur = "USD" Then (
    MajorUnit := "ดอลลาร์";
    MinorUnit := "เซนต์"
) Else If Cur = "CNY" Then (
    MajorUnit := "หยวน";
    MinorUnit := "เฟิน" 
) Else If Cur = "EUR" Then (
    MajorUnit := "ยูโร";
    MinorUnit := "เซนต์"
) Else If Cur = "THB" Or Cur = "" Then (
    MajorUnit := "บาท";
    MinorUnit := "สตางค์"
) Else (
    MajorUnit := Cur; // ถ้าไม่ตรงกับข้างบน ให้แสดงตัวย่อสกุลเงิน
    MinorUnit := "เซนต์"
);

// แยกจำนวนเต็มและทศนิยม
Local StringVar TxtNum := ToText(InputNum, 2, "");
Local StringVar IntPart := Split(TxtNum, ".")[1];
Local StringVar DecPart := Split(TxtNum, ".")[2];

Local StringVar Array TxtNumArr := ["", "หนึ่ง", "สอง", "สาม", "สี่", "ห้า", "หก", "เจ็ด", "แปด", "เก้า"];
Local StringVar Array TxtPosArr := ["", "สิบ", "ร้อย", "พัน", "หมื่น", "แสน", "ล้าน"];

Local StringVar MainText := "";
Local NumberVar i;
Local NumberVar LenInt := Length(IntPart);
Local NumberVar Digit;

// ------------------------------------
// ส่วนที่ 1: คำนวณคำอ่านจำนวนเต็ม
// ------------------------------------
For i := 1 To LenInt Do
(
    Digit := ToNumber(Mid(IntPart, i, 1));
    Local NumberVar Pos := LenInt - i + 1;
    Local NumberVar TruePos := ((Pos - 1) Mod 6) + 1;
    
    If Digit > 0 Then
    (
        If TruePos = 2 And Digit = 2 Then
            MainText := MainText + "ยี่สิบ"
        Else If TruePos = 2 And Digit = 1 Then
            MainText := MainText + "สิบ"
        Else If TruePos = 1 And Digit = 1 And i > 1 Then
            MainText := MainText + "เอ็ด"
        Else
            MainText := MainText + TxtNumArr[Digit + 1] + TxtPosArr[TruePos];
    );
    
    If TruePos = 1 And Pos > 1 Then
        MainText := MainText + "ล้าน";
);

If MainText = "" Or IntPart = "0" Then MainText := "ศูนย์";
MainText := MainText + MajorUnit;

// ------------------------------------
// ส่วนที่ 2: คำนวณคำอ่านทศนิยม (หน่วยย่อย)
// ------------------------------------
Local StringVar SubText := "";
Local NumberVar LenDec := Length(DecPart);

If DecPart = "00" Then
    SubText := "ถ้วน"
Else
(
    For i := 1 To LenDec Do
    (
        Digit := ToNumber(Mid(DecPart, i, 1));
        Local NumberVar PosDec := LenDec - i + 1;
        
        If Digit > 0 Then
        (
            If PosDec = 2 And Digit = 2 Then
                SubText := SubText + "ยี่สิบ"
            Else If PosDec = 2 And Digit = 1 Then
                SubText := SubText + "สิบ"
            Else If PosDec = 1 And Digit = 1 And Left(DecPart, 1) <> "0" Then
                SubText := SubText + "เอ็ด"
            Else If PosDec = 1 And Digit = 1 And Left(DecPart, 1) = "0" Then
                SubText := SubText + "หนึ่ง"
            Else
                SubText := SubText + TxtNumArr[Digit + 1] + TxtPosArr[PosDec];
        );
    );
    SubText := SubText + MinorUnit;
);

// คืนค่าผลลัพธ์
MainText + SubText;
