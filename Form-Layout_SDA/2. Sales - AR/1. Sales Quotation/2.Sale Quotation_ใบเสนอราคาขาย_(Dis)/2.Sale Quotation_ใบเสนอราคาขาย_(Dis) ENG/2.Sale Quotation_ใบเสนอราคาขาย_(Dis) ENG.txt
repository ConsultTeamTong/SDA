// ===== Split integer / decimal =====
nAbs := Abs(nAmount);
nInt := Truncate(nAbs);
nDec := Round((nAbs - nInt) * 100);
If nDec = 100 Then (nInt := nInt + 1; nDec := 0);

nWork := nInt;

// ----- Trillions -----
nGrp  := Truncate(nWork / 1000000000000);
nWork := nWork - nGrp * 1000000000000;
If nGrp > 0 Then (
    nH := Truncate(nGrp / 100);
    nR := nGrp - nH * 100;
    If nH > 0 Then sIntWords := sIntWords + aOnes[nH + 1] + " Hundred ";
    If nR >= 20 Then (
        sIntWords := sIntWords + aTens[Truncate(nR / 10) + 1];
        If nR Mod 10 > 0 Then sIntWords := sIntWords + "-" + aOnes[(nR Mod 10) + 1];
        sIntWords := sIntWords + " "
    ) Else If nR > 0 Then (
        sIntWords := sIntWords + aOnes[nR + 1] + " "
    );
    sIntWords := sIntWords + "Trillion "
);

// ----- Billions -----
nGrp  := Truncate(nWork / 1000000000);
nWork := nWork - nGrp * 1000000000;
If nGrp > 0 Then (
    nH := Truncate(nGrp / 100);
    nR := nGrp - nH * 100;
    If nH > 0 Then sIntWords := sIntWords + aOnes[nH + 1] + " Hundred ";
    If nR >= 20 Then (
        sIntWords := sIntWords + aTens[Truncate(nR / 10) + 1];
        If nR Mod 10 > 0 Then sIntWords := sIntWords + "-" + aOnes[(nR Mod 10) + 1];
        sIntWords := sIntWords + " "
    ) Else If nR > 0 Then (
        sIntWords := sIntWords + aOnes[nR + 1] + " "
    );
    sIntWords := sIntWords + "Billion "
);

// ----- Millions -----
nGrp  := Truncate(nWork / 1000000);
nWork := nWork - nGrp * 1000000;
If nGrp > 0 Then (
    nH := Truncate(nGrp / 100);
    nR := nGrp - nH * 100;
    If nH > 0 Then sIntWords := sIntWords + aOnes[nH + 1] + " Hundred ";
    If nR >= 20 Then (
        sIntWords := sIntWords + aTens[Truncate(nR / 10) + 1];
        If nR Mod 10 > 0 Then sIntWords := sIntWords + "-" + aOnes[(nR Mod 10) + 1];
        sIntWords := sIntWords + " "
    ) Else If nR > 0 Then (
        sIntWords := sIntWords + aOnes[nR + 1] + " "
    );
    sIntWords := sIntWords + "Million "
);

// ----- Thousands -----
nGrp  := Truncate(nWork / 1000);
nWork := nWork - nGrp * 1000;
If nGrp > 0 Then (
    nH := Truncate(nGrp / 100);
    nR := nGrp - nH * 100;
    If nH > 0 Then sIntWords := sIntWords + aOnes[nH + 1] + " Hundred ";
    If nR >= 20 Then (
        sIntWords := sIntWords + aTens[Truncate(nR / 10) + 1];
        If nR Mod 10 > 0 Then sIntWords := sIntWords + "-" + aOnes[(nR Mod 10) + 1];
        sIntWords := sIntWords + " "
    ) Else If nR > 0 Then (
        sIntWords := sIntWords + aOnes[nR + 1] + " "
    );
    sIntWords := sIntWords + "Thousand "
);

// ----- Hundreds (last group) -----
nGrp := nWork;
If nGrp > 0 Then (
    nH := Truncate(nGrp / 100);
    nR := nGrp - nH * 100;
    If nH > 0 Then sIntWords := sIntWords + aOnes[nH + 1] + " Hundred ";
    If nR >= 20 Then (
        sIntWords := sIntWords + aTens[Truncate(nR / 10) + 1];
        If nR Mod 10 > 0 Then sIntWords := sIntWords + "-" + aOnes[(nR Mod 10) + 1]
    ) Else If nR > 0 Then (
        sIntWords := sIntWords + aOnes[nR + 1]
    )
);

If nInt = 0 Then sIntWords := "Zero";
sIntWords := Trim(sIntWords);

// ===== Decimal part =====
If nDec > 0 Then (
    If nDec < 20 Then
        sDecWords := aOnes[nDec + 1]
    Else (
        nTen := Truncate(nDec / 10);
        nOne := nDec - nTen * 10;
        sDecWords := aTens[nTen + 1];
        If nOne > 0 Then sDecWords := sDecWords + "-" + aOnes[nOne + 1]
    )
);

// ===== Build final result =====
If nAmount < 0 Then sNeg := "Minus ";

If bHasMinor Then (
    If nDec > 0 Then
        sResult := sNeg + sIntWords + " " +
                   (If nInt = 1 Then sMajor Else sMajors) +
                   " and " + sDecWords + " " +
                   (If nDec = 1 Then sMinor Else sMinors) + " Only"
    Else
        sResult := sNeg + sIntWords + " " +
                   (If nInt = 1 Then sMajor Else sMajors) + " Only"
) Else (
    sResult := sNeg + sIntWords + " " +
               (If nInt = 1 Then sMajor Else sMajors) + " Only"
);

sResult
